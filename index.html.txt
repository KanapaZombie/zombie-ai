        async function triggerCamera() {
            // On mobile devices, native capture input works best; on desktop, try getUserMedia
            const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent);
            if (isMobile) {
                document.getElementById('camera-input').click();
                return;
            }

            try {
                const stream = await navigator.mediaDevices.getUserMedia({ 
                    video: { facingMode: 'environment' } 
                });
                state.webcamStream = stream;
                const video = document.getElementById('camera-stream');
                video.srcObject = stream;
                video.classList.remove('hidden');
                
                document.getElementById('upload-placeholder').classList.add('hidden');
                document.getElementById('image-preview').classList.add('hidden');
                document.getElementById('capture-controls').classList.add('hidden');
                document.getElementById('webcam-controls').classList.remove('hidden');
            } catch (err) {
                showToast(i18n[state.lang].toastCamError, 'info');
                document.getElementById('camera-input').click();
            }
        }

        async function fetchWithRetry(url, options, maxRetries = 4) {
            let delay = 1000;
            for (let attempt = 0; attempt <= maxRetries; attempt++) {
                try {
                    const response = await fetch(url, options);
                    if (!response.ok) {
                        throw new Error(`HTTP ${response.status}`);
                    }
                    return await response.json();
                } catch (err) {
                    if (attempt === maxRetries) throw err;
                    await new Promise(res => setTimeout(res, delay));
                    delay *= 2;
                }
            }
        }

            const protocolDescriptions = {
                civil_defense: 'EU Civil Protection Mechanism (Decision 1313/2013/EU), Polish Civil Protection & Civil Defense Act (Ustawa o ochronie ludności i obronie cywilnej), MSWiA/PSP Guidelines on Shelters and Places of Temporary Refuge (Miejsca Doraźnego Schronienia - MDS), and RCB (Rządowe Centrum Bezpieczeństwa) preparedness handbooks.',
                fire_evac: 'Polish Fire Protection Act (Ustawa o ochronie przeciwpożarowej), MSWiA Regulation on Fire Protection of Buildings (Dz.U. 2010 nr 109 poz. 719), Państwowa Straż Pożarna (PSP) evacuation route clearance, fire extinguisher, hydrant, and smoke ventilation standards, plus EU fire safety directives.',
                public_infra: 'EU Critical Entities Resilience Directive (CER - Directive (EU) 2022/2557), Polish Crisis Management Act (Ustawa o zarządzaniu kryzysowym), RCB public safety alerts, municipal flood/storm/electrical hazard protocols, and community infrastructure safety.',
                cbrn_hazmat: 'EU CBRN Action Plan, Polish Environmental Protection & GIOŚ (Główny Inspektorat Ochrony Środowiska) standards, PSP Chemical & Ecological Rescue (Ratownictwo Chemiczno-Ekologiczne) protocols, ADR/CLP hazardous materials labeling, and RCB contamination procedures.'
            };

            const systemPrompt = `You are a senior European Union Civil Protection Officer and Polish State Fire Service (Państwowa Straż Pożarna - PSP) & Government Security Center (RCB) Safety Inspector.
Your mission is to analyze the provided photograph for physical, structural, fire, environmental, CBRN, or civil defense hazards affecting civilians and local communities.

CRITICAL INSTRUCTIONS:
1. Respond ENTIRELY in ${targetLanguage}. Every field in the JSON output must be written naturally and accurately in ${targetLanguage}.
2. Focus the inspection through the lens of: ${protocolDescriptions[state.protocol]}.
3. Reference authentic, verified Polish state publications/laws (e.g., KG PSP guidelines, Wytyczne MDS, Rozporządzenie MSWiA ws. ochrony przeciwpożarowej, Poradniki RCB "Bądź gotowy!", Ustawa o ochronie ludności i obronie cywilnej) AND European Union Civil Protection protocols (e.g., EU Civil Protection Mechanism, Union Disaster Resilience Goals, Directive (EU) 2022/2557).
4. Provide practical, life-saving, step-by-step guidance suitable for civilians, building administrators, or local community responders.
5. Output STRICTLY valid JSON matching the requested schema.`;

            const userQuery = `Inspect this image for safety hazards, threats, and civil defense readiness. 
Selected Inspection Framework: ${state.protocol}.
${userNotes ? `User Location/Context Notes: "${userNotes}"` : 'No additional location notes provided.'}
Identify all visible hazards, assess overall risk level (CRITICAL, HIGH, MODERATE, or LOW), provide immediate protective actions for civilians, and cite specific EU Civil Protection and Polish state (PSP/RCB/MSWiA/GOV.PL) standards.`;

            const apiKey = ""; // Provided automatically by runtime environment
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: [
                    {
                        role: "user",
                        parts: [
                            { text: userQuery },
                            {
                                inlineData: {
                                    mimeType: state.imageMimeType,
                                    data: state.imageBase64
                                }
                            }
                        ]
                    }
                ],
                systemInstruction: {
                    parts: [{ text: systemPrompt }]
                },
                tools: [{ "google_search": {} }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "OBJECT",
                        properties: {
                            reportTitle: { type: "STRING", description: "Concise title of the safety inspection report" },
                            riskLevel: { type: "STRING", enum: ["CRITICAL", "HIGH", "MODERATE", "LOW"] },
                            riskLabelLocalized: { type: "STRING", description: "Localized risk badge label, e.g., CRITICAL RISK / KRYTYCZNE ZAGROŻENIE" },
                            executiveSummary: { type: "STRING", description: "Detailed 3-4 sentence assessment of the scene, primary threats, and civil defense readiness" },
                            detectedHazards: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        hazardName: { type: "STRING" },
                                        severity: { type: "STRING", enum: ["CRITICAL", "HIGH", "MODERATE", "LOW"] },
                                        observation: { type: "STRING", description: "What is specifically visible in the photo" },
                                        threatImpact: { type: "STRING", description: "Why this poses a danger to civilians or violates safety rules" }
                                    },
                                    required: ["hazardName", "severity", "observation", "threatImpact"]
                                }
                            },
                            mitigationActions: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        priority: { type: "STRING", description: "e.g., IMMEDIATE / NATYCHMIAST, SHORT-TERM / KRÓTKOTERMINOWE" },
                                        actionTitle: { type: "STRING" },
                                        actionSteps: { type: "STRING" }
                                    },
                                    required: ["priority", "actionTitle", "actionSteps"]
                                }
                            },
                            regulatoryReferences: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        authority: { type: "STRING", description: "e.g., KG PSP / MSWiA, RCB, EU Civil Protection" },
                                        documentTitle: { type: "STRING" },
                                        relevance: { type: "STRING" }
                                    },
                                    required: ["authority", "documentTitle", "relevance"]
                                }
                            }
                        },
                        required: ["reportTitle", "riskLevel", "riskLabelLocalized", "executiveSummary", "detectedHazards", "mitigationActions", "regulatoryReferences"]
                    }
                }
            };

            try {
                const result = await fetchWithRetry(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const candidate = result.candidates?.[0];
                const rawJsonText = candidate?.content?.parts?.[0]?.text;

                if (!rawJsonText) {
                    throw new Error("Empty model response");
                }

                const parsedReport = JSON.parse(rawJsonText);

                // Extract Google Search grounding sources if present
                let sources = [];
                const groundingMetadata = candidate.groundingMetadata;
                if (groundingMetadata && groundingMetadata.groundingAttributions) {
                    sources = groundingMetadata.groundingAttributions
                        .map(attr => ({
                            uri: attr.web?.uri,
                            title: attr.web?.title
                        }))
                        .filter(s => s.uri && s.title);
                }

                state.lastReport = { ...parsedReport, sources };
                renderReport(state.lastReport);

            } catch (error) {
                console.error("Safety Audit Error:", error);
                showToast(i18n[state.lang].toastError, 'error');
            } finally {
                scanOverlay.classList.add('hidden');
                analyzeBtn.disabled = false;
            }
        }

        function renderReport(report) {
            document.getElementById('report-standby').classList.add('hidden');
            const container = document.getElementById('report-container');
            container.classList.remove('hidden');

            // Metadata & Title
            const now = new Date();
            document.getElementById('report-timestamp').textContent = now.toLocaleString(state.lang === 'pl' ? 'pl-PL' : 'en-GB');
            document.getElementById('report-protocol-badge').textContent = state.protocol.replace('_', ' ');
            document.getElementById('report-title').textContent = report.reportTitle;
            document.getElementById('report-summary').textContent = report.executiveSummary;

            // Risk Badge Styling
            const riskBadge = document.getElementById('risk-badge');
            const riskIcon = document.getElementById('risk-icon');
            document.getElementById('risk-label').textContent = report.riskLabelLocalized || report.riskLevel;

            const riskStyles = {
                CRITICAL: 'bg-red-950/90 border-red-500 text-red-300 shadow-red-950/50',
                HIGH: 'bg-orange-950/90 border-orange-500 text-orange-300 shadow-orange-950/50',
                MODERATE: 'bg-amber-950/90 border-amber-500 text-amber-300 shadow-amber-950/50',
                LOW: 'bg-emerald-950/90 border-emerald-500 text-emerald-300 shadow-emerald-950/50'
            };
            riskBadge.className = `px-4 py-2 rounded-xl border font-mono text-xs font-extrabold uppercase tracking-wider flex items-center gap-2 shadow-lg ${riskStyles[report.riskLevel] || riskStyles.MODERATE}`;
            riskIcon.className = report.riskLevel === 'LOW' ? 'fa-solid fa-circle-check' : 'fa-solid fa-triangle-exclamation';

            // Render Detected Hazards
            const hazardsList = document.getElementById('hazards-list');
            hazardsList.innerHTML = '';
            (report.detectedHazards || []).forEach((h, idx) => {
                const badgeColor = h.severity === 'CRITICAL' ? 'text-red-400 bg-red-950/60 border-red-800/60'
                    : h.severity === 'HIGH' ? 'text-orange-400 bg-orange-950/60 border-orange-800/60'
                    : h.severity === 'MODERATE' ? 'text-amber-400 bg-amber-950/60 border-amber-800/60'
                    : 'text-emerald-400 bg-emerald-950/60 border-emerald-800/60';

                const card = document.createElement('div');
                card.className = 'p-4 rounded-xl bg-tactical-950/90 border border-slate-800 hover:border-slate-700 transition';
                card.innerHTML = `
                    <div class="flex items-start justify-between gap-2 mb-1.5">
                        <h4 class="text-sm font-bold text-slate-100">${idx + 1}. ${escapeHtml(h.hazardName)}</h4>
                        <span class="px-2 py-0.5 rounded text-[10px] font-mono font-bold border ${badgeColor}">${escapeHtml(h.severity)}</span>
                    </div>
                    <p class="text-xs text-slate-300 mb-1.5"><strong class="text-slate-400">${state.lang === 'pl' ? 'Obserwacja:' : 'Observation:'}</strong> ${escapeHtml(h.observation)}</p>
                    <p class="text-xs text-slate-400"><strong class="text-slate-400">${state.lang === 'pl' ? 'Skutek / Zagrożenie:' : 'Impact:'}</strong> ${escapeHtml(h.threatImpact)}</p>
                `;
                hazardsList.appendChild(card);
            });

            // Render Mitigation Actions
            const actionsList = document.getElementById('actions-list');
            actionsList.innerHTML = '';
            (report.mitigationActions || []).forEach((act) => {
                const item = document.createElement('div');
                item.className = 'p-3.5 rounded-xl bg-tactical-950/90 border border-slate-800 flex items-start gap-3';
                item.innerHTML = `
                    <div class="mt-0.5 w-6 h-6 rounded-lg bg-emerald-500/15 border border-emerald-500/30 flex items-center justify-center shrink-0 text-emerald-400 text-xs">
                        <i class="fa-solid fa-check"></i>
                    </div>
                    <div class="flex-1">
                        <div class="flex flex-wrap items-center gap-2 mb-1">
                            <span class="text-[10px] font-mono uppercase px-2 py-0.5 rounded bg-slate-800 text-amber-300 font-semibold">${escapeHtml(act.priority)}</span>
                            <h4 class="text-xs sm:text-sm font-bold text-white">${escapeHtml(act.actionTitle)}</h4>
                        </div>
                        <p class="text-xs text-slate-300 leading-relaxed">${escapeHtml(act.actionSteps)}</p>
                    </div>
                `;
                actionsList.appendChild(item);
            });

            // Render Regulatory References
            const protocolsList = document.getElementById('protocols-list');
            protocolsList.innerHTML = '';
            (report.regulatoryReferences || []).forEach((ref) => {
                const box = document.createElement('div');
                box.className = 'p-3.5 rounded-xl bg-tactical-950/90 border border-slate-800 flex flex-col justify-between';
                box.innerHTML = `
                    <div>
                        <span class="inline-block text-[10px] font-mono uppercase text-blue-400 font-bold mb-1">${escapeHtml(ref.authority)}</span>
                        <h4 class="text-xs font-bold text-slate-100 mb-1">${escapeHtml(ref.documentTitle)}</h4>
                    </div>
                    <p class="text-[11px] text-slate-400 leading-snug">${escapeHtml(ref.relevance)}</p>
                `;
                protocolsList.appendChild(box);
            });

            // Render Search Grounding Links if available
            const sourcesWrapper = document.getElementById('grounding-sources-wrapper');
            const sourcesList = document.getElementById('grounding-sources-list');
            sourcesList.innerHTML = '';
            if (report.sources && report.sources.length > 0) {
                sourcesWrapper.classList.remove('hidden');
                // Deduplicate by URI
                const uniqueSources = Array.from(new Map(report.sources.map(s => [s.uri, s])).values()).slice(0, 6);
                uniqueSources.forEach(src => {
                    const a = document.createElement('a');
                    a.href = src.uri;
                    a.target = '_blank';
                    a.rel = 'noopener noreferrer';
                    a.className = 'text-[11px] px-2.5 py-1 rounded-lg bg-slate-800 hover:bg-slate-700 text-blue-300 border border-slate-700 transition flex items-center gap-1.5 max-w-xs truncate';
                    a.innerHTML = `<i class="fa-solid fa-arrow-up-right-from-square text-[9px]"></i><span class="truncate">${escapeHtml(src.title)}</span>`;
                    sourcesList.appendChild(a);
                });
            } else {
                sourcesWrapper.classList.add('hidden');
            }

            // Smooth scroll to report on mobile
            if (window.innerWidth < 1024) {
                container.scrollIntoView({ behavior: 'smooth', block: 'start' });
            }
        }

        function formatReportAsPlainText() {
            if (!state.lastReport) return '';
            const r = state.lastReport;
            let lines = [
                `============================================================`,
                `SCAN4SAFETY PL/EU — CIVIL DEFENSE & HAZARD AUDIT REPORT`,
                `============================================================`,
                `TITLE: ${r.reportTitle}`,
                `RISK LEVEL: ${r.riskLabelLocalized} (${r.riskLevel})`,
                `FRAMEWORK: ${state.protocol.toUpperCase()}`,
                `DATE: ${new Date().toLocaleString()}`,
                ``,
                `--- EXECUTIVE SUMMARY ---`,
                r.executiveSummary,
                ``,
                `--- DETECTED HAZARDS & VIOLATIONS ---`
            ];

            (r.detectedHazards || []).forEach((h, i) => {
                lines.push(`${i + 1}. [${h.severity}] ${h.hazardName}`);
                lines.push(`   Observation: ${h.observation}`);
                lines.push(`   Impact: ${h.threatImpact}`);
            });

            lines.push(``, `--- RECOMMENDED MITIGATION & CIVIL DEFENSE ACTIONS ---`);
            (r.mitigationActions || []).forEach((a, i) => {
                lines.push(`${i + 1}. [${a.priority}] ${a.actionTitle}: ${a.actionSteps}`);
            });

            lines.push(``, `--- APPLICABLE EU DIRECTIVES & POLISH STATE PROTOCOLS ---`);
            (r.regulatoryReferences || []).forEach((ref) => {
                lines.push(`- ${ref.authority} | ${ref.documentTitle}: ${ref.relevance}`);
            });

            lines.push(``, `Emergency Contacts (PL/EU): 112 (General) | 998 (PSP Fire) | 997 (Police) | 999 (Medical)`);
            return lines.join('\n');
        }

        function copyReportToClipboard() {
            const text = formatReportAsPlainText();
            if (!text) return;

            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                showToast(i18n[state.lang].toastCopied, 'info');
            } catch (err) {
                showToast('Copy failed', 'error');
            }
            document.body.removeChild(textarea);
        }

        function downloadReportMarkdown() {
            const text = formatReportAsPlainText();
            if (!text) return;

            const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `Scan4Safety_${state.lang.toUpperCase()}_${new Date().toISOString().slice(0,10)}.txt`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#039;');
        }
